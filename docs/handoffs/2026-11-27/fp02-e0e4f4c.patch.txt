+const names = new Intl.DisplayNames(['zh-CN'], {type: 'currency'});
+for (const code of currencies) {
+  for (const select of [base, quote]) {
+    select.add(new Option(code + ' · ' + names.of(code), code));
+  }
+}
+quote.value = 'CNY';
+
+let pending;
+function clearResult() {
+  pending?.abort();
+  pending = undefined;
+  result.textContent = '';
+  details.textContent = '';
+}
+form.addEventListener('input', clearResult);
+form.addEventListener('submit', async (event) => {
+  event.preventDefault();
+  clearResult();
+  const from = base.value;
+  const to = quote.value;
+  const value = amount.valueAsNumber;
+  if (!Number.isFinite(value) || value < 0) {
+    details.textContent = '请输入有效的非负金额。';
+    return;
+  }
+  if (from === to) {
+    details.textContent = '请选择两种不同的货币。';
+    return;
+  }
+
+  const controller = new AbortController();
+  pending = controller;
+  const timer = setTimeout(() => controller.abort(), 8000);
+  details.textContent = '正在查询参考汇率…';
+  try {
+    const response = await fetch('https://api.frankfurter.dev/v2/rate/' + from + '/' + to, {
+      signal: controller.signal, credentials: 'omit', referrerPolicy: 'no-referrer'
+    });
+    if (!response.ok) throw new Error('汇率服务返回 HTTP ' + response.status);
+    const data = await response.json();
+    if (data.base !== from || data.quote !== to || !/^\d{4}-\d{2}-\d{2}$/.test(data.date)
+        || !Number.isFinite(data.rate) || data.rate <= 0) throw new Error('汇率数据无效');
+    if (pending !== controller) return;
+    const converted = value * data.rate;
+    if (!Number.isFinite(converted)) throw new Error('金额超出换算范围');
+    const format = new Intl.NumberFormat('zh-CN', {style: 'currency', currency: to, currencyDisplay: 'code'});
+    result.textContent = format.format(converted);
+    details.textContent = '1 ' + from + ' = ' + data.rate + ' ' + to + ' · 汇率日期 ' + data.date + ' · 来源 Frankfurter';
+  } catch (error) {

+    details.textContent = controller.signal.aborted ? '汇率查询超时，请重试。' : '汇率查询失败，请重试。' + error.message;
+  } finally {
+    clearTimeout(timer);
+    if (pending === controller) pending = undefined;
+  }
+});
